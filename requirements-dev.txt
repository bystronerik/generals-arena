# Development and test dependencies.
#
#     pip install -r requirements.txt -r requirements-dev.txt

pytest

# Test-only oracle, not a runtime dependency. `tests/test_ratings_model.py`
# checks our damped-Newton Bradley-Terry fit against elote's independently
# written minorization-maximization implementation of the same estimator —
# something no self-consistency test can do. The arena's own rating code does
# not import it.
elote

# Average Joe pipeline: the `joe`-marked test gate (`pytest -m joe`) needs
# the training stack locally; `scripts/joe_modal_*` need the modal client.
# The GPU container pins its own copies inside the Modal image.
equinox
optax
modal

# R2 (S3-compatible) storage for interruptible vast.ai runs:
# `training/joe/store.py` imports it lazily (`R2Store.from_env`), so the
# cheap-suite store tests run against a fake client without it.
boto3

# Build-time only: `arena/bundle.py` minifies each Python member on the way
# into a submission zip. Never installed on the judge and never imported by a
# bot — the zip carries plain source, just rewritten. Pinned because the
# bundle's bytes are its identity: a minifier upgrade would silently change
# every zip we produce.
python-minifier==3.2.0
